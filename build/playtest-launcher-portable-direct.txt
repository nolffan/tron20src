# playtest-launcher-portable-direct.txt -- build/portabletest.sh's second
# run: the modern build's Tron.exe, which starts the game with no launcher
# window (MODERN.md APPS-1), on the prefix of the first run. Its directory
# is a new one, with no tron20.ini: the detail setting is asked for again
# (Medium: 800x600). The game must reach the main menu; then Quit, and
# Tron.exe must stop after the game.
wait 10
screen detail
sclick 400 352
wait 60
screen menu
# Quit (the main menu: the game window is "TRON 2.0", the launcher's
# -windowtitle, at 800x600), then the confirmation
wfocus TRON 30
wclick TRON 270 412
wait 3
screen quit
key Return
wait 15
